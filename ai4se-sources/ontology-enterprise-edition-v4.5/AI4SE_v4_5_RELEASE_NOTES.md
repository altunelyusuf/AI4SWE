
#### **SPARQL**
```sparql
PREFIX : <http://ai4se.org/ontology/v4#>

# Find all portfolios with high ROI
SELECT ?portfolio ?roi ?budget
WHERE {
    ?portfolio a :V4_Portfolio_AIPortfolio ;
               :hasExpectedROI ?roi ;
               :hasTotalBudget ?budget .
    FILTER (?roi > 1.5)
}
ORDER BY DESC(?roi)

# Get all KPI dashboards
SELECT ?dashboard ?kpiCount ?realtime
WHERE {
    ?dashboard a :V4_Performance_KPIDashboard ;
               :hasKPICount ?kpiCount ;
               :isRealTime ?realtime .
}
```
